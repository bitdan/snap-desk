cmake_minimum_required(VERSION 3.24)
project(snap_translator_worker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# 引擎由构建机预先安装，CPU 构建禁用 CUDA；版本与语言包契约必须同步。
find_package(ctranslate2 4.6.0 EXACT CONFIG REQUIRED)
include(FetchContent)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
FetchContent_Declare(sentencepiece GIT_REPOSITORY https://github.com/google/sentencepiece.git GIT_TAG v0.2.0 GIT_SHALLOW TRUE)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(sentencepiece json)
add_executable(snap-translator-worker main.cpp)
target_include_directories(snap-translator-worker PRIVATE ${sentencepiece_SOURCE_DIR}/src)
target_link_libraries(snap-translator-worker PRIVATE CTranslate2::ctranslate2 sentencepiece-static nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(snap-translator-worker PRIVATE /utf-8 /W4)
endif()
# CI 只安装这个组件，避免 FetchContent 子项目的工具、头文件和静态库进入 runtime。
install(TARGETS snap-translator-worker RUNTIME DESTINATION . COMPONENT translator-runtime)
# Windows DLL 必须随程序安装，不能依赖用户开发环境的 PATH。
if(WIN32)
  install(FILES $<TARGET_RUNTIME_DLLS:snap-translator-worker> DESTINATION . COMPONENT translator-runtime)
endif()
